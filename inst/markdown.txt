library(rminka)
library(dplyr)
library(tibble)
library(sf)
library(leaflet)


places <- mnk_places_byname("Forum")
places[,1:6]

place <- mnk_place_sf(253)

place <- place |>
  mutate(
    name = places[2,3],
    id = places[2,1]
  )
place

# 1. descarga y añade columnas

obs_place <- rminka::mnk_place_obs(place_id= 253, year = 2025, month = 2)

obs_place


obs_sf <- obs_place |>
  # 1. quédate solo con lo que necesitas (añade taxon_name si no lo tienes)
  transmute(
    id,
    taxon_name,                    # si no existe, créala antes con mutate(taxon_name = "...")
    observed_on = as.Date(observed_on),
    latitude,
    longitude,
    positional_accuracy
  ) |>
  # 2. elimina filas sin coordenadas
  filter(!is.na(latitude), !is.na(longitude)) |>
  # 3. convierte a puntos WGS84. remove = FALSE mantiene lat/lon como columnas
  st_as_sf(coords = c("longitude", "latitude"), crs = 4326, remove = FALSE)

leaflet() |>
  addProviderTiles("Esri.WorldImagery", group = "Satélite") |>
  addProviderTiles("OpenStreetMap", group = "OSM") |>
  addPolygons(
    data = place,
    color = "#2c7fb8", weight = 2, fillOpacity = 0.1,
    label = ~name
  ) |>
  addMarkers(
    data = obs_sf,
    popup = ~paste0(
      "<b>", taxon_name, "</b><br>",
      "ID: ", id, "<br>",
      "Fecha: ", observed_on
    ),
    clusterOptions = markerClusterOptions()
  ) |>
  addLayersControl(baseGroups = c("Satélite", "OSM"))
